cmake_minimum_required(VERSION 3.31) #this whole file was semi-vibecoded

if(WIN32)
    set(VCPKG_TARGET_TRIPLET "x64-windows-static")
endif()

project(oxygenpad LANGUAGES CXX C)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(VCPKG_MANIFEST_MODE ON)

if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

find_package(SFML 2.5 COMPONENTS graphics window system REQUIRED)
find_package(CURL REQUIRED)

file(GLOB_RECURSE RESOURCE_FILES 
    "${CMAKE_CURRENT_SOURCE_DIR}/src/drawings/*.png"
    "${CMAKE_CURRENT_SOURCE_DIR}/src/fonts/*.ttf"
    "${CMAKE_CURRENT_SOURCE_DIR}/src/sounds/*.wav"
) 

set(RES_CPP_CONTENT "// Auto-generated by CMake. Do not modify.\n#include <map>\n#include <string>\n#include <vector>\n\nnamespace res {\n")

foreach(RES_FILE ${RESOURCE_FILES})
    file(READ "${RES_FILE}" HEX_CONTENT HEX)
    
    file(RELATIVE_PATH REL_PATH "${CMAKE_CURRENT_SOURCE_DIR}" "${RES_FILE}")
    
    string(MAKE_C_IDENTIFIER "${REL_PATH}" VAR_NAME)

    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," C_ARRAY "${HEX_CONTENT}")
    
    string(APPEND RES_CPP_CONTENT "extern const unsigned char ${VAR_NAME}[] = { ${C_ARRAY} 0x00 };\n")
    string(APPEND RES_CPP_CONTENT "extern const unsigned int ${VAR_NAME}_len = sizeof(${VAR_NAME}) - 1;\n\n")
endforeach()

string(APPEND RES_CPP_CONTENT "}\n")

set(GENERATED_RESOURCES_CPP "${CMAKE_CURRENT_BINARY_DIR}/generated_resources.cpp")
file(WRITE "${GENERATED_RESOURCES_CPP}" "${RES_CPP_CONTENT}")

add_library(ass STATIC
    src/logger.h
    src/libs/pfd.h
    src/libs/miniaudio.h
    src/libs/miniaudio.cpp
    src/libs/tray.h
    src/ui.h
    src/ui.cpp
    src/update.h
    src/update.cpp
    "${GENERATED_RESOURCES_CPP}"
) 

set_target_properties(ass PROPERTIES LINKER_LANGUAGE CXX)

target_include_directories(ass PUBLIC 
    ${CMAKE_CURRENT_SOURCE_DIR}
) 

target_link_libraries(ass PUBLIC 
    sfml-graphics
    sfml-window
    sfml-system
    CURL::libcurl
)

function(setup_exe TARGET_NAME SOURCE_FILE)
    set(EXE_SOURCES ${SOURCE_FILE})

    if(WIN32)
        list(APPEND EXE_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/src/resource.rc")
    endif()

    if(WIN32)
        if(CMAKE_BUILD_TYPE STREQUAL "Release")
            add_executable(${TARGET_NAME} WIN32 ${EXE_SOURCES})
            target_link_options(${TARGET_NAME} PRIVATE "/SUBSYSTEM:WINDOWS" "/ENTRY:mainCRTStartup")
        else()
            add_executable(${TARGET_NAME} ${EXE_SOURCES})
        endif()
    else()
        add_executable(${TARGET_NAME} ${EXE_SOURCES})
    endif()
    
    target_link_libraries(${TARGET_NAME} PRIVATE ass)
    target_link_libraries(${TARGET_NAME} PRIVATE sfml-graphics sfml-window sfml-system CURL::libcurl)

    if(WIN32)
        target_link_libraries(${TARGET_NAME} PRIVATE opengl32 winmm gdi32 user32 advapi32 shell32)
    endif()
endfunction()

setup_exe(oxygenpad src/main.cpp)
